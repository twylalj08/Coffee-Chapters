// Coffee & Chapters // Homepage interactions and library data const seasons = [ { name: "Autumn", emoji: "🍂", className: "autumn-theme" }, { name: "Winter", emoji: "❄️", className: "winter-theme" }, { name: "Spring", emoji: "🌸", className: "spring-theme" }, { name: "Summer", emoji: "☀️", className: "summer-theme" } ]; const quotes = [ { text: "A reader lives a thousand lives before they die.", author: "George R. R. Martin" }, { text: "Books are a uniquely portable magic.", author: "Stephen King" }, { text: "There is no friend as loyal as a book.", author: "Ernest Hemingway" }, { text: "Reading is an exercise in empathy.", author: "Malorie Blackman" }, { text: "Some books leave us free and some books make us free.", author: "Ralph Waldo Emerson" } ]; const hadesMessages = [ "Hades is guarding your books.", "Hades demands another chapter.", "Hades has claimed this shelf.", "Hades knocked over a thriller.", "Hades says your TBR is not big enough.", "Hades is pretending not to watch you." ]; const seasonButton = document.getElementById("seasonButton"); const hadesButton = document.getElementById("hadesButton"); const hadesMessage = document.getElementById("hadesMessage"); let currentSeasonIndex = 0; function applySeason(index) { const season = seasons[index]; document.body.classList.remove( "autumn-theme", "winter-theme", "spring-theme", "summer-theme" ); document.body.classList.add(season.className); seasonButton.textContent = `${season.emoji} ${season.name}`; localStorage.setItem("coffeeChaptersSeason", String(index)); } function loadSavedSeason() { const savedSeason = Number( localStorage.getItem("coffeeChaptersSeason") ); if ( Number.isInteger(savedSeason) && savedSeason >= 0 && savedSeason < seasons.length ) { currentSeasonIndex = savedSeason; } applySeason(currentSeasonIndex); } seasonButton.addEventListener("click", () => { currentSeasonIndex = (currentSeasonIndex + 1) % seasons.length; applySeason(currentSeasonIndex); }); hadesButton.addEventListener("click", () => { const randomIndex = Math.floor( Math.random() * hadesMessages.length ); hadesMessage.textContent = hadesMessages[randomIndex]; hadesButton.classList.remove("hades-bounce"); void hadesButton.offsetWidth; hadesButton.classList.add("hades-bounce"); }); function displayDailyQuote() { const today = new Date(); const dayNumber = Math.floor( today.getTime() / 86400000 ); const quote = quotes[dayNumber % quotes.length]; document.getElementById("dailyQuote").textContent = quote.text; document.getElementById("quoteAuthor").textContent = `— ${quote.author}`; } function updateChallenge(finishedCount, goal = 50) { const safeFinished = Number(finishedCount) || 0; const safeGoal = Number(goal) || 50; const percentage = Math.min( Math.round((safeFinished / safeGoal) * 100), 100 ); document.getElementById("challengeProgress").textContent = `${safeFinished} books finished`; document.getElementById("challengeGoal").textContent = `Goal: ${safeGoal}`; document.getElementById("progressFill").style.width = `${percentage}%`; const progressTrack = document.querySelector(".progress-track"); progressTrack.setAttribute( "aria-valuenow", String(percentage) ); const challengeMessage = document.getElementById("challengeMessage"); if (percentage >= 100) { challengeMessage.textContent = "Challenge complete. Absolutely feral reading behaviour."; } else if (percentage >= 75) { challengeMessage.textContent = "The finish line is close. Keep going."; } else if (percentage >= 50) { challengeMessage.textContent = "Halfway there and the shelf is filling up."; } else if (percentage >= 25) { challengeMessage.textContent = "You are building a very respectable stack."; } else { challengeMessage.textContent = "Every finished book brings the shelf to life."; } } function countBooks(data, key) { if (Array.isArray(data[key])) { return data[key].length; } if ( data.stats && Number.isFinite(Number(data.stats[key])) ) { return Number(data.stats[key]); } return 0; } async function loadLibrary() { try { const response = await fetch( "data/library.json", { cache: "no-store" } ); if (!response.ok) { throw new Error( `Library request failed: ${response.status}` ); } const data = await response.json(); const libraryCount = countBooks(data, "library"); const tbrCount = countBooks(data, "tbr"); const finishedCount = countBooks(data, "finished"); document.getElementById("libraryCount").textContent = `${libraryCount} ${libraryCount === 1 ? "Book" : "Books"}`; document.getElementById("tbrCount").textContent = `${tbrCount} ${tbrCount === 1 ? "Book" : "Books"}`; document.getElementById("finishedCount").textContent = `${finishedCount} ${finishedCount === 1 ? "Book" : "Books"}`; if ( data.currentlyReading && data.currentlyReading.title ) { document.getElementById("currentBook").textContent = data.currentlyReading.title; document.getElementById("currentAuthor").textContent = data.currentlyReading.author ? `by ${data.currentlyReading.author}` : "Currently reading"; } else { document.getElementById("currentBook").textContent = "Choose your next adventure..."; document.getElementById("currentAuthor").textContent = "No current book selected"; } const challengeGoal = data.profile && Number(data.profile.readingGoal) ? Number(data.profile.readingGoal) : 50; updateChallenge(finishedCount, challengeGoal); } catch (error) { console.error( "Coffee & Chapters could not load library data:", error ); document.getElementById("currentBook").textContent = "Library data is still brewing..."; document.getElementById("currentAuthor").textContent = "Check data/library.json"; updateChallenge(0, 50); } } loadSavedSeason(); displayDailyQuote(); loadLibrary();
